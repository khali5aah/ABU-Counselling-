4) src/api/http.ts
  export async function request<T>(method: "GET" | "POST", path: string, body?: unknown): Promise<T>
  - URL = config.apiBaseUrl + path (path always starts with "/api/v1/" and ends with "/").
  - Headers: "Content-Type": "application/json", "Accept": "application/json", "ngrok-skip-browser-warning": "true".
  - 15-second timeout via AbortController.
  - fetch throws or aborts → throw new ApiError(status 0, code "network_error", message "Can't reach the server. Check your connection and try again.").
  - 204 → return undefined as T. 2xx → parsed JSON.
  - non-2xx: try to parse JSON; if it has error.code → throw ApiError from the envelope (fields default {}, retryAfter from error.retry_after); otherwise throw ApiError(status, "server_error", "Something went wrong. Please try again.").
