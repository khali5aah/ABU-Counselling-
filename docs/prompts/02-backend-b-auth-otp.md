3) services.py (all business logic lives here; views stay thin)
Helpers:
  hash_code(verification_id, code) = hmac.new(settings.SECRET_KEY.encode(), f"{verification_id}:{code}".encode(), hashlib.sha256).hexdigest()
  hash_setup_token(token) = hashlib.sha256(token.encode()).hexdigest()
  GENERIC_OTP_MESSAGE = "If this number is registered and not yet set up, a verification code has been sent by SMS."

request_otp(phone: str) -> dict          (phone is already normalized by the serializer)
  a) h = phone_hash(phone); now = time.time()
  b) Cooldown: until = cache.get(f"otp:cooldown:{h}"); if until and until > now → raise RateLimitedError(extra={"retry_after": ceil(until - now)})
  c) Hourly cap: win = cache.get(f"otp:window:{h}") or {"start": now, "count": 0}; if win["count"] >= settings.OTP_MAX_REQUESTS_PER_PHONE_PER_HOUR → raise RateLimitedError(extra={"retry_after": max(1, ceil(win["start"] + 3600 - now))})
  d) Record: cache.set(f"otp:cooldown:{h}", now + settings.OTP_RESEND_COOLDOWN_SECONDS, timeout=settings.OTP_RESEND_COOLDOWN_SECONDS); win["count"] += 1; cache.set(f"otp:window:{h}", win, timeout=max(1, int(win["start"] + 3600 - now)))
     (b–d apply to EVERY number, registered or not — this is the anti-enumeration guarantee.)
  e) reg = StudentRegistration.objects.filter(phone_number=phone).first(); eligible = reg is not None and not User.objects.filter(registration=reg).exists()
  f) If not eligible: log INFO "otp request not eligible phone=%s" (masked); return {"request_id": str(uuid4()), "message": GENERIC_OTP_MESSAGE, "expires_in": settings.OTP_TTL_SECONDS, "resend_after": settings.OTP_RESEND_COOLDOWN_SECONDS}  (nothing stored, nothing sent)
  g) If eligible: code = f"{secrets.randbelow(10**6):06d}"; with transaction.atomic(): PhoneVerification.objects.filter(registration=reg, verified_at__isnull=True, invalidated_at__isnull=True).update(invalidated_at=timezone.now()); pv = PhoneVerification(id=uuid4(), registration=reg, phone_number=phone, expires_at=timezone.now()+timedelta(seconds=settings.OTP_TTL_SECONDS), max_attempts=settings.OTP_MAX_ATTEMPTS); pv.code_hash = hash_code(pv.id, code); pv.save()
     AFTER the atomic block: try get_otp_provider().send_code(phone, code) except OtpDeliveryError: log ERROR "otp delivery failed phone=%s" (masked) — do NOT change the response.
     return the same dict shape as (f) with request_id = str(pv.id).

verify_otp(phone: str, request_id: UUID, code: str) -> dict
  IMPORTANT: raising an exception inside transaction.atomic() rolls back the attempts counter. Compute an outcome inside the atomic block, save, LEAVE the block, then raise.
  with transaction.atomic():
    pv = PhoneVerification.objects.select_for_update().filter(id=request_id, phone_number=phone).first()
    if pv is None or pv.invalidated_at or pv.verified_at: outcome = ("invalid", None)
    elif pv.attempts >= pv.max_attempts: outcome = ("attempts", None)
    elif timezone.now() >= pv.expires_at: outcome = ("expired", None)
    elif not hmac.compare_digest(pv.code_hash, hash_code(pv.id, code)):
        pv.attempts += 1; pv.save(update_fields=["attempts"]); remaining = pv.max_attempts - pv.attempts
        outcome = ("attempts", None) if remaining <= 0 else ("wrong", remaining)
    else:
        token = secrets.token_urlsafe(32); pv.verified_at = now; pv.setup_token_hash = hash_setup_token(token); pv.setup_token_expires_at = now + timedelta(seconds=settings.SETUP_TOKEN_TTL_SECONDS); pv.save(); outcome = ("ok", token)
  then: "invalid" → raise InvalidOtpError(); "wrong" → raise InvalidOtpError(extra={"attempts_remaining": remaining}); "attempts" → OtpAttemptsExceededError(); "expired" → OtpExpiredError(); "ok" → return {"setup_token": token, "expires_in": settings.SETUP_TOKEN_TTL_SECONDS}

setup_account(setup_token: str, username: str, password: str) -> User      (username already normalized by UsernameField)
  Order matters (contract: validation errors and duplicate username do NOT consume the token):
  with transaction.atomic():
    pv = PhoneVerification.objects.select_for_update().select_related("registration").filter(setup_token_hash=hash_setup_token(setup_token)).first()
    if pv is None or pv.setup_token_used_at or pv.setup_token_expires_at is None or timezone.now() >= pv.setup_token_expires_at: outcome = "token_invalid"
    elif User.objects.filter(registration=pv.registration).exists(): pv.setup_token_used_at = now; pv.save(); outcome = "token_invalid"
    else:
      try: validate_password(password, user=User(username=username)) except DjangoValidationError as e: outcome = ("password", list(e.messages))
      if no outcome and User.objects.filter(username=username).exists(): outcome = "duplicate"
      if no outcome:
        try: with transaction.atomic(): user = User.objects.create_user(username=username, password=password, registration=pv.registration)
        except IntegrityError: outcome = "duplicate" if User.objects.filter(username=username).exists() else "token_invalid"
        else: pv.setup_token_used_at = now; pv.save(update_fields=["setup_token_used_at"]); outcome = ("ok", user)
  then: "token_invalid" → SetupTokenInvalidError(); ("password", msgs) → raise serializers.ValidationError({"password": msgs}); "duplicate" → DuplicateError(fields={"username": ["This username is taken."]}); ok → return user

issue_tokens(user) -> dict: refresh = RefreshToken.for_user(user); return {"access": str(refresh.access_token), "refresh": str(refresh), "user": UserSerializer(user).data}
